##*****     Bien doi Box-Cox

##
##--------------------------------------------------------
##
## Cach 1: Viet ham 
##         + Viet ham tinh profile-lig likelihood ;
##         + Tim gia tri lambda toi uu; 
##         + Viet ham thuc hien bien doi Box-Cox;
##
##
## Ham thuc hien bien doi Box-Cox:
##
boxcox_trans <- function(x, lambda){
  if(lambda == 0) log(x)
  else (x^lambda - 1)/lambda
}
##
##------------------------------------------------------------------------------
## Ham tinh profile-log likelihood
##
pl <- function(lambda, formula, data){
  y <- model.extract(model.frame(formula, data = data), "response")
  y_bcx <- boxcox_trans(x = y, lambda = lambda)
  data$y_bcx <- y_bcx
  new_formula <- update(formula, y_bcx ~ . )
  out_lm <- lm(new_formula, data = data)
  pl_res <- -0.5*length(y)*log(mean(resid(out_lm)^2)) + (lambda - 1)*sum(log(y))
  return(pl_res)
}

##
##
##------------------------------------------------------------------------------
## Viet ham tinh khoang tin cay
##
pl_interval <- function(formula, data, out_pl, ci.lev){
  ff <- function(lambda, formula, data, max.vals, ci.lev){
    pl_res <- pl(lambda = lambda, formula = formula, data = data)
    out <- pl_res - max.vals + 0.5*qchisq(ci.lev, 1)
    return(out)
  }
  lambda_l <- uniroot(f = ff, interval = c(-2, out_pl$maximum), formula = formula,
                      data = data, max.vals = out_pl$objective, ci.lev = ci.lev)$root
  lambda_u <- uniroot(f = ff, interval = c(out_pl$maximum, 2), formula = formula, 
                      data = data, max.vals = out_pl$objective, ci.lev = ci.lev)$root
  return(c(lambda_l, lambda_u))
}
##
##------------------------------------------------------------------------------
## 
## Tim khaong tin cay 95% cho lambda
ci_lambda <- pl_interval(formula = time ~ treat + poison, data = poisons, out_pl = out_optim_pl,
                         ci.lev = 0.95)
ci_lambda
##

plot(x_lambda, pl_est, type = "l", xlab = "lambda", ylab = "profile-log likelihood")
abline(v = ci_lambda, lty = 2)
